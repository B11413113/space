/* USER CODE BEGIN 4 */
void HAL_TIM_PeriodElapsedCallback(TIM_HandleTypeDef *htim)
{
  if (htim->Instance == TIM6)
  {
    // 1. 根據當前波形狀態輸出對應的靜態查表數值到 DAC
    uint16_t val = 2048;
    switch (current_wave) {
        case WAVE_SINE:
            val = sine_table[wave_index];
            break;
        case WAVE_TRIANGLE:
            val = triangle_table[wave_index];
            break;
        case WAVE_SQUARE:
            val = square_table[wave_index];
            break;
        default:
            val = sine_table[wave_index];
            break;
    }
    
    HAL_DAC_SetValue(&hdac1, DAC_CHANNEL_1, DAC_ALIGN_12B_R, val);
    
    // 索引遞增 (100 點循環)
    wave_index++;
    if (wave_index >= 100)
    {
      wave_index = 0;
    }

    // 2. 按鍵掃描與消抖 (每 5ms 檢查一次，5kHz 下計數 25 次)
    static uint16_t poll_cnt = 0;
    poll_cnt++;
    if (poll_cnt >= 25)
    {
      poll_cnt = 0;
      // PC13 在按下時為 Low (0)，放開為 High (1)
      uint8_t raw_state = (HAL_GPIO_ReadPin(GPIOC, GPIO_PIN_13) == GPIO_PIN_RESET) ? 0 : 1;
      
      static uint8_t last_raw_state = 1;
      static uint8_t stable_state = 1;
      static uint8_t debounce_count = 0;

      if (raw_state == last_raw_state)
      {
        debounce_count++;
        if (debounce_count > 4) // 超過約 20ms 狀態穩定
        {
          if (stable_state != raw_state)
          {
            stable_state = raw_state;
            // 抓取下降緣 (按下瞬間) 切換波形
            if (stable_state == 0)
            {
              current_wave = (WaveType_t)((current_wave + 1) % WAVE_MAX);
            }
          }
        }
      }
      else
      {
        debounce_count = 0;
        last_raw_state = raw_state;
      }
    }
  }
}
