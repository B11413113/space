int main(void)
{

  HAL_TIM_PWM_Start(&htim3, TIM_CHANNEL_3);

  /* Infinite loop */
  /* USER CODE BEGIN WHILE */
  /* USER CODE BEGIN WHILE */

  /* USER CODE BEGIN WHILE */
    while (1)
    {
      /* 1. 時間間隔縮短至 1 毫秒 (最快更新頻率)，確保細節最滑順 */
      if (HAL_GetTick() - last_tick >= 1)
      {
        last_tick = HAL_GetTick(); // 更新時間標記

        /* 2. 增加一個延遲計數器，每 N 毫秒才真正更新一次步距，讓低亮度區停留更久 */
        delay_cnt++;
        if (delay_cnt >= 5) // 如果覺得還是太快，可以把 5 改大 (例如 10)，越大學暗亮切換越久
        {
            delay_cnt = 0; // 重置計數器

            /* 3. 線性調整原始步進值 (0 到 1000) */
            raw_step += step_dir;

            // 當原始步進值達到上限或下限時，反轉變化方向
            if (raw_step >= 1000)
            {
              raw_step = 1000;
              step_dir = -1; // 改為遞減 (變暗)
            }
            else if (raw_step <= 0)
            {
              raw_step = 0;
              step_dir = 1;  // 改為遞增 (變亮)
            }

            /* 4. [關鍵核心] 將原始線性值經過非線性公式 (平方曲線) 轉換，做出符合視覺的亮度 */
            // 公式解釋: final_duty = (raw_step / 1000)^2 * 999
            // 我們利用 (raw_step * raw_step) 做出平方項，再除以 (1000*1000) 的比例因子，
            // 最後乘以 ARR 上限 999。
            final_duty = (raw_step * raw_step * 999) / 1000000;

            /* 5. 將計算好的非線性、極滑順數值寫入 TIM3 Channel 3 */
            __HAL_TIM_SET_COMPARE(&htim3, TIM_CHANNEL_3, final_duty);
        }
      }
      /* USER CODE END WHILE */

      /* USER CODE BEGIN 3 */
    }
    /* USER CODE END 3 */
}
